''' app_py = r'''import os import logging from flask import Flask, request, jsonify, send_from_directory from dotenv import load_dotenv from groq import Groq BASE_DIR = os.path.dirname(os.path.abspath(__file__)) load_dotenv(os.path.join(BASE_DIR, ".env")) app = Flask(__name__, static_folder=BASE_DIR) logging.basicConfig(level=logging.INFO) api_key = os.getenv("GROQ_API_KEY") client = Groq(api_key=api_key) if api_key else None SYSTEM_PROMPT = """ You are CareerMudhra Assistant. Answer factual questions about colleges, courses, fees, eligibility, and admissions in a concise, neutral way. Important limits: - You do not have access to CareerMudhra's website content or a verified knowledge base in this demo unless information is included in the user's message. - Do not claim that you checked CareerMudhra pages or official sources. - Never invent college fees, eligibility criteria, cutoffs, deadlines, seat availability, rankings, or admission guarantees. - If you cannot reliably answer from your available knowledge, say that the detail needs to be checked on CareerMudhra's website or the relevant college/official authority. - Do not provide personalized counselling or recommend which college a person should choose. - Do not request sensitive personal information. - Distinguish general information from details that must be verified. Add GROQ_API_KEY to the .env file and restart app.py." }), 500 body = request.get_json(silent=True) or {} question = body.get("message", "") if not isinstance(question, str) or not question.strip(): return jsonify({"error": "Please enter a question."}), 400 question = question.strip() if len(question) > 2000: return jsonify({"error": "Please keep your question under 2,000 characters."}), 400 try: result = client.chat.completions.create( model="openai/gpt-oss-120b", messages=[ {"role": "system", "content": SYSTEM_PROMPT}, {"role": "user", "content": question} ], max_tokens=700, temperature=0.2 ) answer = result.choices[0].message.content or "" return jsonify({"answer": answer.strip()}) except Exception: # Keep provider details and credentials out of the browser response. app.logger.exception("Groq API request failed") return jsonify({ "error": "The AI provider request failed. Check your API key, model access, network connection, and Groq usage limits; then try again." }), 502 if __name__ == "__main__": # Local demo only. Use a proper production server and HTTPS before public deployment. app.run(host="127.0.0.1", port=5000, debug=False) ''' requirements = """Flask>=3.0,<4.0 groq>=0.9.0 python-dotenv>=1.0 """ env_example = """# Copy this file to .env and paste your Groq API key after the equals sign.
